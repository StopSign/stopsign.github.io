# The words "slap" and "sandal" mean TWO different things in this codebase:
#
#   A) INTRUDER RANGE (how enemies fight)
#      - mapData.js:  createZone(count, "scoundrel", ...)  or  createZone(..., "lecher", ...)
#      - maps.js:     intruderType: intruderType || "scoundrel"
#      - engine.js:   intruder.intruderType === "lecher", lecherReady, slapQueue (queue of
#                     intruders in reach — NOT your slap)
#      Theme: close creeps = scoundrels, distant = lechers
#      These tags describe UNIT RANGE, not the Madame's attacks.
#
#   B) MADAME COMBAT (player slap / sandal throw)
#      - sandalAttack, baseSlapDamage, slapFlatBeforeStr, getEffectiveSlapCooldownMs

#      - engine.js:  pushAttackLineFx(target, "slap"), __OVERKILL__BoostKind: "slap"
#      Theme: slap (close), sandal throw (far)
#

# wrongly rewrite createZone(5, "scoundrel", ...) into createZone(5, "slap", ...) and merge
# two unrelated ideas.
